cmake_minimum_required(VERSION 3.20)
project(stillpoint LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(stillpoint engine/main.cpp)
target_compile_options(stillpoint PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(name IN ITEMS ledger crash crash_write bounds divide loop arguments)
  add_executable(${name} examples/${name}.cpp)
  target_compile_options(${name} PRIVATE -g -O0 -fno-omit-frame-pointer -fno-pie)
  target_link_options(${name} PRIVATE -no-pie -Wl,--build-id=sha1)
endforeach()

add_executable(pie_crash examples/crash.cpp)
target_compile_options(pie_crash PRIVATE -g -O0 -fno-omit-frame-pointer -fPIE)
target_link_options(pie_crash PRIVATE -pie -Wl,--build-id=sha1)
